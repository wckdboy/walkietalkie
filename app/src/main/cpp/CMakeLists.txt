cmake_minimum_required(VERSION 3.22)
project(walkietalkie LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(RUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/arm64-v8a")
foreach(LIB llama ggml ggml-base)
 add_library(${LIB} SHARED IMPORTED)
 set_target_properties(${LIB} PROPERTIES IMPORTED_LOCATION "${RUNTIME_DIR}/lib${LIB}.so")
endforeach()
add_library(walkietalkie SHARED walkietalkie.cpp)
target_include_directories(walkietalkie PRIVATE ../../../../third_party/llama-include)
target_link_libraries(walkietalkie llama ggml ggml-base android log dl)
target_link_options(walkietalkie PRIVATE "-Wl,-z,max-page-size=16384")
add_library(walkietalkie-speech SHARED speech.cpp)
target_link_libraries(walkietalkie-speech dl)
target_link_options(walkietalkie-speech PRIVATE "-Wl,-z,max-page-size=16384")
